    wt = randn(5,1).*Rwi;     % generate random noise

function dx = dxstm(x,A,B,u)
    dx = A*x + B*u;
end

function dk = Kdot(K,A,B,R,Q)
    k = reshape(K,size(A));
    kdot = -k*A + k*B*inv(R)*B'*k - Q - A'*k;
    dk = kdot(:);
end

function ds = Sdot(s,t,K,A,B,R,Q,tk)
   % r = [0; 0; sin(t*pi/5); 0; 0];
   k = interp1(tk,K,t);
   s = reshape(s',5,5,[]);

   ds = -(A' - k*B*inv(R)*B')*s + Q*[0; 0; sin(t*pi/5); 0; 0];
end
